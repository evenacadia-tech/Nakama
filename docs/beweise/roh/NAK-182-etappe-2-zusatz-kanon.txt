Dein Kanonstart blieb an der Ask-Regel fuer das Loeschen der alten Logdatei haengen (die Regel in .claude/settings.json bleibt bestehen; der Dirigent gibt nichts frei, er nennt den Ersatzweg).   $log = Join-Path $env:TEMP 'nakama-nak182-kanon.log'
  $befehl = "pwsh -NoProfile -File tools/beweise.ps1 -Bauen -Ziel docs/beweise/NAK-182.md -Anhaengen -Titel NAK-182 *> $log; Add-Content $log ('EXIT=' + `$LASTEXITCODE)"
  Start-Process pwsh -WindowStyle Hidden -WorkingDirectory (Get-Location) -ArgumentList '-NoProfile','-Command',$befehl
