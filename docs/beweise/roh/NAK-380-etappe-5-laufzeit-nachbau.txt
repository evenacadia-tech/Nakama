    import json, struct, numpy as np
    FIX = r"C:\Users\phili\Projekte\Nakama\eq-copilot\schemas\v3\bandgitter\nakama_1_24_oct_30_18k_v1.json"
    fein = json.load(open(FIX, encoding="utf-8"))
    h2d = lambda s: struct.unpack(">d", bytes.fromhex(s))[0]
    K = np.array([h2d(s) for s in fein["kanten_hz"]["hex64"]]); MIT = np.array([h2d(s) for s in fein["mitten_hz"]["hex64"]])
    rng = np.random.default_rng(3)
    N = 4096; hop = 2048; w = 0.5 - 0.5 * np.cos(2 * np.pi * np.arange(N) / N)
    fs = 48000.0; d = 48; frames = 22
    trenn = next(b for b in range(221) if MIT[b] >= 200.0)
    baender = []
    for b in range(trenn, 221):
        if K[b + 1] > 18000.0: continue
        v = int(np.ceil(K[b] * N / fs)); bi = min(int(np.ceil(K[b + 1] * N / fs)), N // 2 + 1)
        if bi - v >= 2: baender.append((b, v, bi))
    fehler = {b: [] for b, _, _ in baender}
    for lauf in range(40):
        n = N + (frames - 1) * hop
        x = rng.uniform(-1, 1, n + d)
        l = x[d:d + n]; r = x[0:n]
        S = np.zeros(N // 2 + 1, complex)
        for f in range(frames):
            s = f * hop
            S += np.fft.rfft(l[s:s + N] * w) * np.conj(np.fft.rfft(r[s:s + N] * w))
        df = fs / N
        for b, v, bi in baender:
            lag = np.sum(S[v + 1:bi] * np.conj(S[v:bi - 1]))
            fehler[b].append(np.angle(lag) / (2 * np.pi * df) * 1000.0 - 1000.0 * d / fs)
    alle = np.array([np.mean(v) for v in fehler.values()])
    sd = np.array([np.std(v) for v in fehler.values()])
    bins = np.array([bi - v for _, v, bi in baender])
    print("Mittel ueber Baender und Laeufe %.5f ms, SD der Bandmittel %.5f" % (alle.mean(), alle.std()))
    for nb in (2, 3, 5, 10, 20, 39):

        if sel.any():
            print("Bins %2d: Bias %.4f ms, SD je Lauf %.4f ms (Baender %d)" % (nb, alle[sel].mean(), sd[sel].mean(), sel.sum()))
