function Wert($StatementAst) {
    # Der Wert eines Schluessels, wenn er eine konstante Zeichenkette ist, sonst $null.
    if ($StatementAst -isnot [System.Management.Automation.Language.PipelineAst]) { return $null }

        $prozess = Start-Process @start
        $null = $prozess.Handle
        $prozess.WaitForExit()
        $uhr.Stop()
        $code = $prozess.ExitCode
        $sekunden = [Math]::Round($uhr.Elapsed.TotalSeconds, 2).ToString($Invariant)
        $text = [IO.File]::ReadAllText($aus, [Text.Encoding]::UTF8)
        $fehlertext = [IO.File]::ReadAllText($err, [Text.Encoding]::UTF8)
        $zeilen = @($text.TrimEnd("`r", "`n") -split "`r?`n")
        $wieErwartet = ($code -eq $l.Soll)
        if (-not $wieErwartet) { $befunde++ }
        N ''
        N ('### ' + $l.K + ' - ' + $befehl)
        N ('Beginn ' + $beginn.ToString('HH:mm:ss') + ', Dauer ' + $sekunden + ' s, Exit ' + $code + ' (erwartet ' + $l.Soll + ') - ' + $(if ($wieErwartet) { 'wie erwartet' } else { 'ABWEICHUNG' }) + ', ' + $zeilen.Count + ' Ausgabezeilen')
        N 'Traegerzeilen der nachgezogenen Behauptung:'
        foreach ($t in @($zeilen | Where-Object { $_ -match $l.Traeger })) { N ('  ' + $t) }
        N 'Letzte Zeilen:'
        for ($i = [Math]::Max(0, $zeilen.Count - 12); $i -lt $zeilen.Count; $i++) { N ('  ' + $zeilen[$i]) }
        if ($fehlertext.Trim()) {
            N 'stderr:'
            foreach ($t in ($fehlertext.TrimEnd() -split "`r?`n")) { N ('  ' + $t) }
        }
        $zaehl = @($zeilen | Where-Object { $_ -match 'Pruefungen ok|geprueft, .* Fehler|bestanden, .* gescheitert|^Selbsttest: |TEST (OK|FEHLGESCHLAGEN)|GOLDEN (OK|FEHLGESCHLAGEN)|^URTEIL' }) -join ' / '
        $tabelle.Add(('| {0} | {1} | {2} ({3}) | {4} s | {5} |' -f $l.K, $befehl, $code, $l.Soll, $sekunden, $zaehl))
    }

    N ''
    N '## 5. Zusammenfassung'
    N ''
    N '| Bein | Befehl | Exit (erwartet) | Dauer | Zaehlzeilen |'
    N '|---|---|---|---|---|'
    foreach ($s in $tabelle) { N $s }
    N ''
    N ('Gesamturteil: ' + $(if ($befunde -eq 0) { 'OK - keine Parse-Fehler, genau sechs Behauptungen geaendert, Binaries frisch, alle sechs Beine mit erwartetem Exit' } else { 'BEFUND (' + $befunde + ')' }))
    N ''
    N $Marke
    N ''
    foreach ($t in ((SkriptText) -split "`r?`n")) { N $t }
    [IO.File]::WriteAllText($Ausgabe, (($z -join "`n") + "`n"), (New-Object System.Text.UTF8Encoding $false))
    Write-Host ('Rohausgabe: ' + $Ausgabe + ' - Befunde ' + $befunde)
    exit $(if ($befunde -eq 0) { 0 } else { 1 })
}

# Die gefahrenen Binaries gegen den Baustand des Zwischenkanons: derselbe
# SHA-256-Praefix heisst, dass hier dieselben Bytes liefen, die der Kanon
# als frisch bestaetigt hat.
$kanonRoh = [IO.File]::ReadAllText((Join-Path $Wurzel 'docs\beweise\roh\NAK-283-e6771fc-dirty.md'), [Text.Encoding]::UTF8)
$hashOk = $true
N 'Binaries gegen den Baustand des Zwischenkanons (docs/beweise/roh/NAK-283-e6771fc-dirty.md, SHA-256-Praefix mit 16 Zeichen):'
foreach ($ziel in @('EqCopProbeeqNullTest', 'EqCopTransactionTest', 'EqCopDspGoldenTest', 'EqCopMarkierungTest', 'EqCopSonde013TruePeakGoldenTest')) {
    $exe = Join-Path $Wurzel ('eq-copilot\build\plugin\{0}_artefacts\Release\{0}.exe' -f $ziel)
    $ist = (Get-FileHash -LiteralPath $exe -Algorithm SHA256).Hash.Substring(0, 16)
    $treffer = [regex]::Match($kanonRoh, '(?m)^\| `' + [regex]::Escape($ziel) + '` \| ([^|]+) \| `([0-9A-F]{16})` \| ([^|]+) \|')
    $baustand = if ($treffer.Success) { $treffer.Groups[2].Value } else { '(nicht im Baustand)' }
    $hashGleich = $treffer.Success -and ($baustand -eq $ist)
    if (-not $hashGleich) { $hashOk = $false }
    N ('  {0}: ist {1}, Baustand {2} (gebaut {3}, {4}) - {5}' -f $ziel, $ist, $baustand, $(if ($treffer.Success) { $treffer.Groups[1].Value.Trim() } else { '-' }), $(if ($treffer.Success) { $treffer.Groups[3].Value.Trim() } else { '-' }), $(if ($hashGleich) { 'gleich' } else { 'ABWEICHUNG' }))
}

$manifest = [IO.File]::ReadAllText((Join-Path $Wurzel 'docs\beweise\NAK-283.md'), [Text.Encoding]::UTF8)
$a0 = $manifest.IndexOf('### 38.2 ')
$a1 = $manifest.IndexOf('## Kanon-Lauf - NAK-283')
if ($a0 -lt 0 -or $a1 -lt $a0) { throw 'Abschnitt 38.2 steht nicht vor dem Runner-Anhang' }
$abschnitt = $manifest.Substring($a0, $a1 - $a0)
$bloecke = @([regex]::Matches($abschnitt, '(?ms)^```text\r?\n(.*?)\r?\n```') | ForEach-Object { $_.Groups[1].Value })
$bAlt = Beine (Parse (BasisKopie)).Ast
$bNeu = Beine (Parse (Join-Path $Wurzel 'tools\beweise.ps1')).Ast
$soll = New-Object System.Collections.Generic.List[string]
foreach ($k in $Geaendert) {
    $soll.Add((Eintrag $bAlt $k).Behauptung)
    $soll.Add((Eintrag $bNeu $k).Behauptung)
}
N ('Zitatabgleich Abschnitt 38.2: ' + $bloecke.Count + ' text-Bloecke, erwartet ' + $soll.Count + ' (je Bein alt und neu, Reihenfolge ' + ($Geaendert -join ', ') + ')')
$abweichend = 0
for ($i = 0; $i -lt $soll.Count; $i++) {
    $k = $Geaendert[[int][Math]::Floor($i / 2)]
    $art = if ($i % 2 -eq 0) { 'alt (Basis-SHA)' } else { 'neu (Arbeitsbaum)' }
    $woertlich = ($i -lt $bloecke.Count) -and ($bloecke[$i] -ceq $soll[$i])
    if (-not $woertlich) { $abweichend++ }
    N ('  ' + $k + ' ' + $art + ': ' + $(if ($woertlich) { 'woertlich gleich' } else { 'ABWEICHUNG' }))
}
$nachtragOk = ($dokuExit -eq 0) -and $hashOk -and ($abweichend -eq 0) -and ($bloecke.Count -eq $soll.Count)
N ('Urteil Nachtrag: ' + $(if ($nachtragOk) { 'OK' } else { 'BEFUND' }))

$neu = $vorne + ($z -join "`n") + "`n`n" + $Marke + "`n`n" + (SkriptText) + "`n"
[IO.File]::WriteAllText($Ausgabe, $neu, (New-Object System.Text.UTF8Encoding $false))
Write-Host ('Nachtrag: ' + $(if ($nachtragOk) { 'OK' } else { 'BEFUND' }))
exit $(if ($nachtragOk) { 0 } else { 1 })
